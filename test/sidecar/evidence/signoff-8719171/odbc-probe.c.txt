#include <stdio.h>
#include <sql.h>
int main(void) {
    SQLHENV env = SQL_NULL_HENV;
    SQLRETURN result = SQLAllocHandle(SQL_HANDLE_ENV, SQL_NULL_HANDLE, &env);
    printf("SQLAllocHandle=%d\n", (int)result);
    if (!SQL_SUCCEEDED(result)) return 1;
    result = SQLFreeHandle(SQL_HANDLE_ENV, env);
    printf("SQLFreeHandle=%d\n", (int)result);
    return SQL_SUCCEEDED(result) ? 0 : 1;
}
